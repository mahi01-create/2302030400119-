import numpy as np
import pygame
import sys

# Initialize pygame
pygame.init()

# Constants
WIDTH, HEIGHT = 500, 500
CELL_SIZE = 20
FPS = 10

# Colors
WHITE = (255, 255, 255)
GREEN = (0, 200, 0)
RED = (200, 0, 0)
BLACK = (0, 0, 0)

# Set up display
screen = pygame.display.set_mode((WIDTH, HEIGHT))
pygame.display.set_caption("Snake Game using NumPy")
clock = pygame.time.Clock()

# Directions as numpy arrays
UP = np.array([0, -CELL_SIZE])
DOWN = np.array([0, CELL_SIZE])
LEFT = np.array([-CELL_SIZE, 0])
RIGHT = np.array([CELL_SIZE, 0])

# Initialize snake
snake = np.array([[100, 100], [80, 100], [60, 100]])
direction = RIGHT

# Initialize food
def generate_food():
    return np.random.randint(0, WIDTH // CELL_SIZE) * CELL_SIZE, np.random.randint(0, HEIGHT // CELL_SIZE) * CELL_SIZE

food = np.array(generate_food())

def draw_snake():
    for segment in snake:
        pygame.draw.rect(screen, GREEN, (segment[0], segment[1], CELL_SIZE, CELL_SIZE))

def game_over():
    pygame.quit()
    sys.exit()

while True:
    screen.fill(BLACK)

    # Handle events
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            game_over()
        if event.type == pygame.KEYDOWN:
            if event.key == pygame.K_UP and not np.array_equal(direction, DOWN):
                direction = UP
            if event.key == pygame.K_DOWN and not np.array_equal(direction, UP):
                direction = DOWN
            if event.key == pygame.K_LEFT and not np.array_equal(direction, RIGHT):
                direction = LEFT
            if event.key == pygame.K_RIGHT and not np.array_equal(direction, LEFT):
                direction = RIGHT

    # Move snake
    new_head = snake[0] + direction
    snake = np.vstack([new_head, snake[:-1]])

    # Check collision with wall
    if (new_head[0] < 0 or new_head[0] >= WIDTH or
        new_head[1] < 0 or new_head[1] >= HEIGHT):
        game_over()

    # Check collision with self
    if any((new_head == snake[1:]).all(axis=1)):
        game_over()

    # Check if snake eats food
    if np.array_equal(new_head, food):
        snake = np.vstack([new_head, snake])  # grow snake
        food = np.array(generate_food())

    # Draw food
    pygame.draw.rect(screen, RED, (food[0], food[1], CELL_SIZE, CELL_SIZE))

    # Draw snake
    draw_snake()

    pygame.display.update()
    clock.tick(FPS)
